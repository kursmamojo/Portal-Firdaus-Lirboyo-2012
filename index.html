Hubungkan ke Google Sheets Panitia

Kirim otomatis setiap pengisian formulir langsung ke Google Spreadsheet Anda

URL Google Apps Script Panitia telah terpasang secara bawaan. Setiap kali alumni menekan tombol Simpan & Kirim, data langsung dikirim dan disimpan secara realtime ke Google Spreadsheet Anda.

Cara Instan Tanpa Coding (Copy-Paste Langsung):

Buka tab 2. Spreadsheet Data, klik tombol "Salin ke Google Sheets (TSV)", lalu buka file Google Sheets kosong dan tekan Ctrl + V. Seluruh 16 kolom akan langsung tertata rapi!

Script Otomatis Google Sheets (Code.gs)

Tempelkan kode ini di menu Extensions > Apps Script pada Google Sheets Anda

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : e.parameter;
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Timestamp", "ID Alumni", "Nama Lengkap & Gelar", "Nama Panggilan",
      "No. WhatsApp", "Email", "Provinsi Domisili", "Kab/Kota Domisili",
      "Kontak Darurat", "Kamar/Asrama/Madrasah", "Tahun Mondok",
      "Profesi Utama", "Nama Instansi/Usaha", "Bidang Keahlian",
      "Konfirmasi Kehadiran", "Pendamping Keluarga", "Catatan Khusus"
    ]);
  }
  
  sheet.appendRow([
    new Date(), data.id, data.namaLengkap, data.namaPanggilan,
    "'" + data.noWhatsapp, data.emailAktif, data.provinsiDomisili,
    data.kabKotaDomisili, data.kontakDarurat, data.kamarAsrama,
    data.tahunMondok, data.profesiUtama, data.namaInstansi,
    data.bidangKeahlian, data.konfirmasiKehadiran,
    data.jumlahPendamping, data.catatanKhusus
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: []}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var result = rows.slice(1).map(function(r, idx) {
    return {
      timestamp: String(r[0] || ""),
      id: String(r[1] || ("FRD-2012-" + ("000" + (idx + 1)).slice(-3))),
      namaLengkap: String(r[2] || ""),
      namaPanggilan: String(r[3] || ""),
      noWhatsapp: String(r[4] || "").replace(/^'/, ""),
      emailAktif: String(r[5] || ""),
      provinsiDomisili: String(r[6] || ""),
      kabKotaDomisili: String(r[7] || ""),
      kontakDarurat: String(r[8] || ""),
      kamarAsrama: String(r[9] || ""),
      tahunMondok: String(r[10] || ""),
      profesiUtama: String(r[11] || ""),
      namaInstansi: String(r[12] || ""),
      bidangKeahlian: String(r[13] || "").split(",").map(function(s){return s.trim();}).filter(Boolean),
      konfirmasiKehadiran: String(r[14] || "Hadir"),
      jumlahPendamping: String(r[15] || "Sendiri (0)"),
      catatanKhusus: String(r[16] || "")
    };
  });
  return ContentService.createTextOutput(JSON.stringify({status: "success", data: result}))
    .setMimeType(ContentService.MimeType.JSON);
}
